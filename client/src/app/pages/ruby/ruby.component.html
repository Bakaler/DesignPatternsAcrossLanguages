oop — module interface & service pattern
# module as interface
module UserRepository
  def find_by_id(id)
    raise NotImplementedError
  end

  def save(user)
    raise NotImplementedError
  end

  def delete(id)
    raise NotImplementedError
  end
end
# concrete implementation
class PostgresUserRepo
  include UserRepository

  def initialize(conn)
    @conn = conn
  end

  def find_by_id(id)
    @conn.exec_params(
      "SELECT * FROM users WHERE id=$1",
      [id]
    ).first
  end
end
# value object + service
User = Struct.new(:id, :email, :role,
                   keyword_init: true)

class UserService
  def initialize(repo)
    @repo = repo
  end

  def get_or_raise(id)
    @repo.find_by_id(id) ||
      raise(NotFoundError, id)
  end
end

# keyword arguments with defaults
def find_user(id, repo: @repo, cache: true)
  cache ? @cache[id] : repo.find_by_id(id)
end

# lambda middleware
auth = ->(req) {
  raise UnauthorizedError unless req.token?

# yield to block
def with_connection
  conn = DB.connect
  yield conn
ensure
  conn&.close
end
