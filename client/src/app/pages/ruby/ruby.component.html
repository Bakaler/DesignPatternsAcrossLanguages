oop — module interface & service pattern
# module as interface
module UserRepository
  def find_by_id(id)
    raise NotImplementedError
  end

  def save(user)
    raise NotImplementedError
  end

  def delete(id)
    raise NotImplementedError
  end
end
# concrete implementation
class PostgresUserRepo
  include UserRepository

  def initialize(conn)
    @conn = conn
  end

  def find_by_id(id)
    @conn.exec_params(
      "SELECT * FROM users WHERE id=$1",
      [id]
    ).first
  end
end
# value object + service
User = Struct.new(:id, :email, :role,
                   keyword_init: true)

class UserService
  def initialize(repo)
    @repo = repo
  end

  def get_or_raise(id)
    @repo.find_by_id(id) ||
      raise(NotFoundError, id)
  end
end

RSpec.describe Calculator do
  subject(:calc) { described_class.new }

  before(:each) { calc.reset }
  after(:each)  { }
  before(:all)  { # one-time setup }

  it "adds two numbers" do
    expect(calc.add(2, 3)).to eq(5)
  end

  context "when dividing by zero" do
    it "raises ZeroDivisionError" do
      expect { calc.divide(10, 0) }
        .to raise_error(ZeroDivisionError)
    end
  end

  # data-driven
  [[1,1,2],[0,5,5],[-1,1,0]].each do |a,b,want|
    it "add({a},{b}) = {want}" do
      expect(calc.add(a,b)).to eq(want)
    end
  end
end

# let — lazy memoised helper
RSpec.describe UserService do
  let(:repo)   { double("Repo") }
  let(:mailer) { spy("Mailer") }
  let(:svc)    { described_class
                   .new(repo, mailer) }

  it "sends welcome email on create" do
    user = User.new("alice")
    allow(repo).to receive(:save)
    svc.create(user)
    expect(mailer).to have_received(:send)
      .with(user.email, anything)
  end
end
mocking & test doubles
# double — simple fake
repo = double("UserRepository")
allow(repo).to receive(:find_by_id)
             .with(1)
             .and_return(User.new("alice"))
allow(repo).to receive(:find_by_id)
             .with(99)
             .and_raise(NotFoundError)
allow(repo).to receive(:save)

# instance_double — verifies interface
repo = instance_double(UserRepository)

# spy — record without setup
mailer = spy("Mailer")
svc.notify("alice@example.com")
expect(mailer).to
  have_received(:send).once
# expect — assert AND stub in one
expect(repo).to receive(:save)
               .with(an_instance_of(User))
               .once

# verify call count
expect(mailer).to
  have_received(:send).exactly(2).times
expect(mailer).not_to
  have_received(:send)

# partial stub — mock built-in
allow(Time).to receive(:now)
            .and_return(Time.new(2024,1,1))

# stub chained calls
allow(request).to receive_message_chain(
  :headers, :authorization)
  .and_return("Bearer token123")
# integration — real database
RSpec.describe UserRepository,
               type: :integration do
  let(:db)   { Database.connect(":memory:") }
  let(:repo) { described_class.new(db) }

  before(:all)  { db.create_tables }
  after(:each) { db.clear }
  after(:all)  { db.close }

  it "saves and retrieves a user" do
    repo.save(User.new(id: 1, name: "Alice"))
    found = repo.find_by_id(1)
    expect(found.name).to eq("Alice")
  end
end
