oop - service & repository pattern
// interface
public interface IOrderRepository {
    Task<Order?> FindByIdAsync(Guid id);
    Task<IEnumerable<Order>> FindByUserAsync(Guid userId);
    Task SaveAsync(Order order);
}
// EF Core implementation
public class EfOrderRepository : IOrderRepository {
    private readonly AppDbContext _db;

    public async Task<Order?> FindByIdAsync(Guid id)
        => await _db.Orders.FindAsync(id);

    public async Task SaveAsync(Order o) {
        _db.Orders.Add(o);
        await _db.SaveChangesAsync();
    }
}
// record - immutable API response
public record OrderResponse(
    Guid Id,
    string Status,

);

[HttpGet("{id}")]
public async Task<IActionResult> GetOrder(Guid id) {

    return order is null ? NotFound() : Ok(order);
}

// async service method
public async Task<Order> PlaceOrderAsync(OrderRequest req) {
    var user = await _users.FindByIdAsync(req.UserId);
    var order = Order.Create(user, req.Items);
    await _repo.SaveAsync(order);
    await _events.PublishAsync(new OrderPlaced(order.Id));
    return order;
}
