    [Fact]
    public void Push_IncreasesCount() {
        _stack.Push(1);
        Assert.Equal(1, _stack.Count);
    }

    public void Pop_OnEmpty_Throws() {
        Assert.Throws<InvalidOperationException>(
            () => _stack.Pop());

    [InlineData(1)]
    [InlineData(5)]
    [InlineData(10)]

        _stack.Push(val);
        Assert.Equal(val, _stack.Pop());
    }

         .Throws<NotFoundException>();

// argument matching
_mockRepo.Setup(r =>
    r.FindById(It.IsAny<int>()))
    .Returns((int id) => new User($"u{id}"));

    It.IsAny<User>()))
    .Callback<User>(u => saved = u);
// integration — file system
[Fact]
public async Task
    FileLogger_WritesToDisk()
{
    var dir = Path.Combine(
        Path.GetTempPath(),
        Guid.NewGuid().ToString());
    Directory.CreateDirectory(dir);
    try {
        var logger = new FileLogger(dir);
        await logger.LogAsync("hello");
        var content = await
            File.ReadAllTextAsync(
                Path.Combine(dir, "app.log"));
        Assert.Contains("hello", content);
    } finally {
        Directory.Delete(dir, true);
    }
}

using Microsoft.Playwright;

[Test]
public async Task LoginFlow()
{
    using var pw =
        await Playwright.CreateAsync();
    var browser = await
        pw.Chromium.LaunchAsync();
    var page = await
        browser.NewPageAsync();

    await page.GotoAsync("/login");
    await page.FillAsync(
        "[name=email]",
        "alice@example.com");
    await page.ClickAsync("[type=submit]");
    await page.WaitForURLAsync(
        "**/dashboard");
    Assert.AreEqual("Welcome",
        await page.Locator("h1")
                  .TextContentAsync());
}
