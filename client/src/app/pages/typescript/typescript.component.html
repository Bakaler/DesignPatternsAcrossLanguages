    save(user: User): Promise<void>;
    delete(id: UserId): Promise<void>;

// Postgres implementation
class PgUserRepository implements IUserRepository {
    constructor(private db: Pool) {}

    async findById(id: UserId): Promise<User | null> {
        const row = await this.db.query(
            `SELECT * FROM users WHERE id = $1`, [id]
        );
        return row.rows[0] ?? null;
    }
}
// abstract base - shared query logic
abstract class BaseRepository<T> {
    constructor(protected db: Pool) {}

    abstract tableName(): string;

    async findAll(): Promise<T[]> {
        const res = await this.db.query(
            `SELECT * FROM ${this.tableName()}`
        );
        return res.rows;
    }
}

// generic HTTP client
async function get<T>(url: string): Promise<T> {
    const res = await fetch(url);
    if (!res.ok) throw new HttpError(res.status);
    return res.json() as Promise<T>;
}

functions - middleware & handlers
// typed request handler
type Handler = (req: Request, res: Response) => Promise<void>;

// middleware factory
function withAuth(handler: Handler): Handler {
    return async (req, res) => {
        const token = req.headers["authorization"];
        if (!token) return res.status(401).json({ error: "Unauthorized" });
        await handler(req, res);
    };
}

type guards
// narrow DB error vs unknown
function isDbError(e: unknown): e is DbError {
    return e instanceof Error && "code" in e;
}

try {
    await db.query(sql);

concepts
unit — single function or class, all deps faked integration — real dependencies, no browser e2e — full user flow through a real browser mock — controlled fake replacing a dependency stub — forces a return value or throws an error spy — records calls while delegating to real logic fake — lightweight in-memory implementation test suite — describe block grouping related tests coverage — % of lines and branches exercised AAA — Arrange · Act · Assert TDD — write the failing test before the code test pyramid — many unit · some integration · few e2e

mocking & test doubles
// fake — full in-memory stand-in
class FakeUserRepo
  implements UserRepository {
  private db = new
    Map<number, User>();

  async findById(id: number) {
    return this.db.get(id) ?? null;
  }
  async save(user: User) {
    this.db.set(user.id, user);
  }
}

it('round-trips a user', async () => {
  const repo = new FakeUserRepo();
  const svc  = new UserService(repo);
  await svc.create({ id: 1,
    name: 'alice' });
  assert.strictEqual(
    (await svc.get(1)).name, 'alice');
});
// spy — record calls, delegate behaviour
const calls: unknown[][] = [];
const spyRepo: UserRepository = {
  async findById(id: number) {
    calls.push(['findById', id]);
    return { id, name: 'alice' };
  },
  async save(user: User) {
    calls.push(['save', user]);
  },
};

await new UserService(spyRepo)
  .create({ id: 1, name: 'bob' });

assert.strictEqual(calls.length, 1);
assert.strictEqual(calls[0][0], 'save');
assert.strictEqual(
  (calls[0][1] as User).name, 'bob');
// stub — hard-coded return / throw
const stubRepo: UserRepository = {

};

it('throws when not found', async () => {
  await assert.rejects(
    () => new UserService(stubRepo)
              .getOrThrow(99));

// integration — real file system
import fs   from 'node:fs/promises';
import os   from 'node:os';
import path from 'node:path';

describe('FileLogger', () => {
  let dir: string;

  before(async () => {
    dir = await fs.mkdtemp(
      path.join(os.tmpdir(), 'test-'));
  });
  after(async () =>
    fs.rm(dir, { recursive: true }));

  it('writes to disk', async () => {
    await new FileLogger(dir)
      .log('hello world');
    const txt = await fs.readFile(
      path.join(dir, 'app.log'), 'utf-8');
    assert.ok(txt.includes('hello world'));
  });
});
// integration — real HTTP server
import http from 'node:http';

describe('GET /health', () => {
  let server: http.Server;
  let url: string;

  before((done) => {
    server = app.listen(0, () => {
      const addr = server.address() as any;
      url = 'http://localhost:' + addr.port;
      done();
    });
  });
  after((done) => server.close(done));

  it('returns 200', async () => {
    const res = await fetch(url + '/health');
    assert.strictEqual(res.status, 200);
    const b = await res.json();
    assert.strictEqual(b.status, 'ok');
  });
});
