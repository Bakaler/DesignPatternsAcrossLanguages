public:
    virtual ~Repository() = default;
    virtual User  findById(int id) = 0; // pure virtual
    virtual void save(const User& u) = 0;

    std::string connStr;

        : connStr{std::move(conn)} {}

    void save(const User& u) override;
private:
    void connect();
};

templates
// function template
template<typename T>
T clamp(T val, T lo, T hi) {
    return val < lo ? lo : val > hi ? hi : val;
}

auto p = clamp(8080, 1024, 65535); // int
// class template — typed cache
template<typename K, typename V>
class Cache {
public:
    void set(const K& k, V v) {
        store[k] = std::move(v);
    }
    std::optional<V> get(const K& k) const;
private:
    std::unordered_map<K, V> store;
};
// concept constraint (C++20)
template<typename T>
concept Serialisable =
    requires(T t) {
        { t.toJson() } -> std::convertible_to<std::string>;
    };

template<Serialisable T>
void send(const T& payload) {
    transmit(payload.toJson());
}

error handling
// custom exception hierarchy
struct AppError : std::runtime_error {
    int code;
    AppError(int c, std::string m)
        : std::runtime_error{std::move(m)}, code{c} {}
};
struct NotFoundError : AppError {
    NotFoundError(std::string m) : AppError{404, std::move(m)} {}
};

try {
    auto user = repo.findById(id);
} catch (const NotFoundError& e) {
    respond(e.code, e.what());
} catch (const std::exception& e) {
    respond(500, e.what());
}

useful keywords
override — enforce virtual override final — prevent subclassing explicit — block implicit conversion noexcept — guarantee no throw constexpr — compile-time evaluation inline — hint for inlining mutable — allow mutation in const method static — class-level or file-local delete — disable copy/move default — compiler-generated special member [[nodiscard]] — warn on ignored return if constexpr — compile-time branch
