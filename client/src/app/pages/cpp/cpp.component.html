// function template
template<typename T>
T clamp(T val, T lo, T hi) {
    return val < lo ? lo : val > hi ? hi : val;
}

// class template — typed cache
template<typename K, typename V>
class Cache {
public:
    void set(const K& k, V v) {
        store[k] = std::move(v);
    }
    std::optional<V> get(const K& k) const;
private:
    std::unordered_map<K, V> store;
};
// concept constraint (C++20)
template<typename T>
concept Serialisable =
    requires(T t) {
        { t.toJson() } -> std::convertible_to<std::string>;
    };

template<Serialisable T>
void send(const T& payload) {
    transmit(payload.toJson());
}

error handling
// custom exception hierarchy
struct AppError : std::runtime_error {
    int code;
    AppError(int c, std::string m)
        : std::runtime_error{std::move(m)}, code{c} {}
};
struct NotFoundError : AppError {
    NotFoundError(std::string m) : AppError{404, std::move(m)} {}
};

try {
    auto user = repo.findById(id);
} catch (const NotFoundError& e) {
    respond(e.code, e.what());
} catch (const std::exception& e) {
    respond(500, e.what());
}

// integration — real file system
class FileLoggerTest
    : public testing::Test {
protected:
    fs::path tmpDir;
    void SetUp() override {
        tmpDir = fs::temp_directory_path()
               / "test_logger";
        fs::create_directory(tmpDir);
    }
    void TearDown() override {
        fs::remove_all(tmpDir);
    }
};

TEST_F(FileLoggerTest, WritesToDisk) {
    FileLogger log(tmpDir / "app.log");
    log.write("hello world");
    std::ifstream f(tmpDir / "app.log");
    std::string content{
        std::istreambuf_iterator<char>(f),{}};
    EXPECT_THAT(content,
        HasSubstr("hello world"));
}
